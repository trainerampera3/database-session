--program to find the users who has given the most ratings
with after_joining as ( select * from MovieRating join Users on MovieRating.user_id=Users.user_id),
counter as (select count(*) as count,user_id from after_joining group by user_id),
ranker as (select user_id,DENSE_RANK()  over( order by count DESC) as rank from counter)

select * from ranker where rank=1;






WITH banned_users AS (
    SELECT users_id
    FROM Users
    WHERE banned = 'Yes'
),
unbanned_users AS (
    SELECT *
    FROM Trips
    WHERE client_id NOT IN (
        SELECT users_id
        FROM banned_users
    )
    AND driver_id NOT IN (
        SELECT users_id
        FROM banned_users
    )
)
SELECT
request_at,
    SUM(
        CASE
            WHEN status IN ('cancelled_by_driver', 'cancelled_by_client')
            THEN 1
            ELSE 0
        END
    ) AS cancelled_count,
count(*)
FROM unbanned_users group by request_at;
